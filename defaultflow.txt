/* ======================= PSF Debug: compile-time controls ======================= */
/*
 * These controls apply to the default RX-flow debug path that observes frames
 * after CPSW DMA reception.
 *
 * Terminology note:
 * - Here, PHY Status Frames (PSFs) are always observed on the host RX side, even when
 *   they report timestamps caused by outbound traffic.
 *
 * Debug groups:
 * - RX_FRAME_DEBUG:
 *   periodic overview/statistics logging for received traffic classification
 * - RX_FRAME_DUMP:
 *   optional hexdump of selected received frames, intended only for short
 *   format-inspection runs
 * - RX_DROP_STATS:
 *   periodic CPSW drop/error counter logging
 *
 * Echo control:
 * - DROP_PHY_STATUS_FROM_ECHO:
 *   keeps PHY Status Frames in the RX path so they can still be observed and
 *   forwarded into the PHY ingest API, but prevents the example's generic echo
 *   logic from putting those same PSFs back onto the wire.
 */

static uint16_t EnetApp_getEthType(const uint8_t *frame,
                                   uint32_t frameLen,
                                   uint32_t *l2HdrLen)
{
    uint16_t ethType = 0U;
    uint32_t hdrLen = 14U;

    if ((frame == NULL) || (frameLen < hdrLen))
    {
        if (l2HdrLen != NULL)
        {
            *l2HdrLen = 0U;
        }
        return 0U;
    }

    ethType = ((uint16_t)frame[12U] << 8U) | frame[13U];
    while (((ethType == 0x8100U) || (ethType == 0x88A8U)) &&
           (frameLen >= (hdrLen + 4U)))
    {
        hdrLen += 4U;
        ethType = ((uint16_t)frame[hdrLen - 2U] << 8U) | frame[hdrLen - 1U];
    }

    if (l2HdrLen != NULL)
    {
        *l2HdrLen = hdrLen;
    }

    return ethType;
}

static uint8_t EnetApp_getPtpMsgType(const uint8_t *frame,
                                     uint32_t frameLen,
                                     uint32_t l2HdrLen,
                                     uint16_t *seqId)
{
    uint8_t msgType = 0xFFU;

    if ((frame != NULL) && (frameLen > l2HdrLen))
    {
        msgType = frame[l2HdrLen] & 0x0FU;
    }

    if (seqId != NULL)
    {
        *seqId = 0xFFFFU;
        if ((frame != NULL) && (frameLen >= (l2HdrLen + 32U)))
        {
            *seqId = ((uint16_t)frame[l2HdrLen + 30U] << 8U) |
                     frame[l2HdrLen + 31U];
        }
    }

    return msgType;
}

static const char *EnetApp_getPtpMsgTypeName(uint8_t msgType)
{
    switch (msgType)
    {
        case 0x0U: return "sync";
        case 0x1U: return "delay_req";
        case 0x2U: return "pdelay_req";
        case 0x3U: return "pdelay_resp";
        case 0x8U: return "follow_up";
        case 0x9U: return "delay_resp";
        case 0xAU: return "pdelay_resp_fup";
        case 0xBU: return "announce";
        case 0xCU: return "signaling";
        case 0xDU: return "management";
        default:   return "other";
    }
}

#if ENETAPP_ENABLE_RX_DROP_STATS
/* ======================= PSF Debug: drop-stat visibility ======================= */
static void EnetApp_printRxDropStats(void)
{
    static uint32_t lastPrintTimeMs = 0U;
    const uint32_t nowMs = ClockP_getTimeUsec() / 1000U;
    CpswStats_HostPort_Ng hostStats;
    Enet_IoctlPrms prms;
    int32_t status = ENET_SOK;

    if (lastPrintTimeMs == 0U)
    {
        lastPrintTimeMs = nowMs;
        return;
    }
    if ((nowMs - lastPrintTimeMs) < ENETAPP_RX_DROP_STATS_INTERVAL_MS)
    {
        return;
    }

    memset(&hostStats, 0, sizeof(hostStats));
    ENET_IOCTL_SET_OUT_ARGS(&prms, &hostStats);
    ENET_IOCTL(gEnetAppCfg.hEnet, gEnetAppCfg.coreId, ENET_STATS_IOCTL_GET_HOSTPORT_STATS, &prms, status);
    if (status == ENET_SOK)
    {
        EnetAppUtils_print(
            "%s: RX drop stats[1]: aleDrop=%llu aleOverrun=%llu rxBot=%llu rxTop=%llu portMask=%llu rateLimit=%llu\r\n",
            gEnetAppCfg.name,
            hostStats.aleDrop,
            hostStats.aleOverrunDrop,
            hostStats.rxBottomOfFifoDrop,
            hostStats.rxTopOfFifoDrop,
            hostStats.portMaskDrop,
            hostStats.aleRateLimitDrop);
        EnetAppUtils_print(
            "%s: RX drop stats[2]: vid=%llu block=%llu secure=%llu auth=%llu lenErr=%llu ipNxt=%llu ipv4Frag=%llu\r\n",
            gEnetAppCfg.name,
            hostStats.aleVidIngressDrop,
            hostStats.aleBlockDrop,
            hostStats.aleSecureDrop,
            hostStats.aleAuthDrop,
            hostStats.aleLenErrorDrop,
            hostStats.aleIpNextHdrDrop,
            hostStats.aleIPv4FragDrop);
    }
    else
    {
        EnetAppUtils_print("%s: failed to get host drop stats: %d\r\n", gEnetAppCfg.name, status);
    }

    lastPrintTimeMs = nowMs;
}

}

/* ======================= PSF Debug: PTP RX default callback ======================= */
void rxDefaultDataCb(void *data, int size, int port, void *cbArg)
{
#if ENETAPP_ENABLE_PTP_PATH_DEBUG
    static uint32_t cbTraceTotal = 0U;
    static uint32_t cbTracePhyStatus = 0U;
    static uint32_t cbTracePtpNonPsf = 0U;
    static uint32_t cbTraceOther = 0U;
    static uint32_t cbTraceFirstNLogs = 0U;
    static uint32_t cbTraceLastPrintTimeMs = 0U;
    static const uint8_t phyStatusHdr[14U] = {
        0x01, 0x1B, 0x19, 0x00, 0x00, 0x00,
        0x08, 0x00, 0x17, 0x0B, 0x6B, 0x0F,
        0x88, 0xF7,
    };
    const uint8_t *frame = (const uint8_t *)data;
    const uint32_t frameLen = (uint32_t)size;
    const uint32_t nowMs = ClockP_getTimeUsec() / 1000U;
    uint32_t l2HdrLen = 0U;
    uint16_t ethType = 0U;

    cbTraceTotal++;
    if ((frameLen >= sizeof(phyStatusHdr)) &&
        (memcmp(frame, phyStatusHdr, sizeof(phyStatusHdr)) == 0))
    {
        cbTracePhyStatus++;
    }
    else
    {
        ethType = EnetApp_getEthType(frame, frameLen, &l2HdrLen);
        if (ethType == 0x88F7U)
        {
            uint16_t seqId = 0xFFFFU;
            uint8_t msgType = EnetApp_getPtpMsgType(frame, frameLen, l2HdrLen, &seqId);

            cbTracePtpNonPsf++;
            if (cbTraceFirstNLogs < ENETAPP_PTP_PATH_DEBUG_FIRST_N)
            {
                cbTraceFirstNLogs++;
                EnetAppUtils_print(
                    "%s: shared-rx default-cb PTP[%u] port=%d len=%u dst=%02x:%02x:%02x:%02x:%02x:%02x src=%02x:%02x:%02x:%02x:%02x:%02x type=%s(0x%02x) seqId=%u\r\n",
                    gEnetAppCfg.name,
                    cbTraceFirstNLogs,
                    port,
                    frameLen,
                    frame[0U], frame[1U], frame[2U], frame[3U], frame[4U], frame[5U],
                    frame[6U], frame[7U], frame[8U], frame[9U], frame[10U], frame[11U],
                    EnetApp_getPtpMsgTypeName(msgType),
                    msgType,
                    seqId);
            }
        }
        else
        {
            cbTraceOther++;
        }
    }

    if (cbTraceLastPrintTimeMs == 0U)
    {
        cbTraceLastPrintTimeMs = nowMs;
    }
    else if ((nowMs - cbTraceLastPrintTimeMs) >= ENETAPP_RX_FRAME_DEBUG_PRINT_INTERVAL_MS)
    {
        EnetAppUtils_print(
            "%s: shared-rx default-cb summary: total=%u phyStatus=%u ptpNonPsf=%u other=%u\r\n",
            gEnetAppCfg.name,
            cbTraceTotal,
            cbTracePhyStatus,
            cbTracePtpNonPsf,
            cbTraceOther);
        cbTraceLastPrintTimeMs = nowMs;
    }
#endif

#if ENETAPP_ENABLE_RX_FRAME_DEBUG
    static uint32_t total = 0U;
    static uint32_t phyStatus = 0U;
    static uint32_t ptp = 0U;
    static uint32_t ipv4 = 0U;
    static uint32_t arp = 0U;
    static uint32_t ipv6 = 0U;
    static uint32_t lldp = 0U;
    static uint32_t tsn = 0U;
    static uint32_t vlanTagged = 0U;
    static uint32_t other = 0U;
    static uint32_t lastPrintTimeMs = 0U;
    static uint32_t sampleDecimCounter = 0U;
#if ENETAPP_ENABLE_RX_FRAME_DUMP
    static uint32_t dumpDecimCounter = 0U;
    static uint32_t lastDumpTimeMs = 0U;
#endif
    uint16_t ethType = 0U;
    uint32_t l2HdrLen = 14U;
    uint32_t dumpBucket = 0U;
    const char *bucketName = "other";
    const uint8_t *frame = (const uint8_t *)data;
    const uint32_t frameLen = (uint32_t)size;
    const uint32_t nowMs = ClockP_getTimeUsec() / 1000U;
    uint32_t sampleEveryN = ENETAPP_RX_FRAME_DEBUG_SAMPLE_EVERY_N_FRAMES;
    static const uint8_t phyStatusHdr[14U] = {
        0x01, 0x1B, 0x19, 0x00, 0x00, 0x00,
        0x08, 0x00, 0x17, 0x0B, 0x6B, 0x0F,
        0x88, 0xF7,
    };

    (void)cbArg;
    (void)port;

    if (sampleEveryN == 0U)
    {
        sampleEveryN = 1U;
    }

    sampleDecimCounter++;
    if ((frameLen >= 14U) && (sampleDecimCounter >= sampleEveryN))
    {
        sampleDecimCounter = 0U;
        total++;
        if (memcmp(frame, phyStatusHdr, sizeof(phyStatusHdr)) == 0)
        {
            phyStatus++;
            dumpBucket = ENETAPP_RX_DUMP_BUCKET_PHY_STATUS;
            bucketName = "phyStatus";
        }
        else
        {
            ethType = ((uint16_t)frame[12U] << 8U) | frame[13U];
            if ((ethType == 0x8100U) || (ethType == 0x88A8U))
            {
                vlanTagged++;
                dumpBucket = ENETAPP_RX_DUMP_BUCKET_VLAN;
                bucketName = "vlan";
                l2HdrLen += 4U;
                if (frameLen >= l2HdrLen)
                {
                    ethType = ((uint16_t)frame[l2HdrLen - 2U] << 8U) | frame[l2HdrLen - 1U];
                }
            }
            if ((ethType == 0x8100U) || (ethType == 0x88A8U))
            {
                vlanTagged++;
                dumpBucket = ENETAPP_RX_DUMP_BUCKET_VLAN;
                bucketName = "vlan";
                l2HdrLen += 4U;
                if (frameLen >= l2HdrLen)
                {
                    ethType = ((uint16_t)frame[l2HdrLen - 2U] << 8U) | frame[l2HdrLen - 1U];
                }
            }
            if (ethType == 0x88F7U)
            {
                ptp++;
                dumpBucket = ENETAPP_RX_DUMP_BUCKET_PTP;
                bucketName = "ptp";
            }
            else if (ethType == 0x0800U)
            {
                ipv4++;
                dumpBucket = ENETAPP_RX_DUMP_BUCKET_IPV4;
                bucketName = "ipv4";
            }
            else if (ethType == 0x0806U)
            {
                arp++;
            }
            else if (ethType == 0x86DDU)
            {
                ipv6++;
            }
            else if (ethType == 0x88CCU)
            {
                lldp++;
            }
            else if (ethType == 0x22F0U)
            {
                tsn++;
            }
            else
            {
                other++;
                dumpBucket = ENETAPP_RX_DUMP_BUCKET_OTHER;
                bucketName = "other";
            }
        }

        if (lastPrintTimeMs == 0U)
        {
            lastPrintTimeMs = nowMs;
        }
        else if ((nowMs - lastPrintTimeMs) >= ENETAPP_RX_FRAME_DEBUG_PRINT_INTERVAL_MS)
        {
            EnetAppUtils_print(
                "%s: PTP-RX flow overview: total=%u phyStatus=%u ptp=%u ipv4=%u arp=%u ipv6=%u lldp=%u tsn=%u vlan=%u other=%u\r\n",
                gEnetAppCfg.name, total, phyStatus, ptp, ipv4, arp, ipv6, lldp, tsn, vlanTagged, other);
            lastPrintTimeMs = nowMs;
        }

#if ENETAPP_ENABLE_RX_FRAME_DUMP
        {
            uint32_t dumpDecimation = ENETAPP_RX_FRAME_DUMP_DECIMATION;
            uint32_t maxDumpBytes = ENETAPP_RX_FRAME_DUMP_MAX_BYTES;
            uint32_t i;

            if (dumpDecimation == 0U)
            {
                dumpDecimation = 1U;
            }
            dumpDecimCounter++;
            if (maxDumpBytes > frameLen)
            {
                maxDumpBytes = frameLen;
            }

            if ((maxDumpBytes > 0U) &&
                ((ENETAPP_RX_FRAME_DUMP_BUCKET_MASK & dumpBucket) != 0U) &&
                (dumpDecimCounter >= dumpDecimation) &&
                ((lastDumpTimeMs == 0U) || ((nowMs - lastDumpTimeMs) >= ENETAPP_RX_FRAME_DUMP_INTERVAL_MS)))
            {
                dumpDecimCounter = 0U;
                lastDumpTimeMs = nowMs;
                EnetAppUtils_print("%s: PTP-RX dump bucket=%s len=%u bytes=%u\r\n",
                                   gEnetAppCfg.name, bucketName, frameLen, maxDumpBytes);
                for (i = 0U; i < maxDumpBytes; i++)
                {
                    if ((i % 16U) == 0U)
                    {
                        EnetAppUtils_print("  %03u:", i);
                    }
                    EnetAppUtils_print(" %02x", frame[i]);
                    if ((((i + 1U) % 16U) == 0U) || ((i + 1U) == maxDumpBytes))
                    {
                        EnetAppUtils_print("\r\n");
                    }
                }
            }
        }
#endif
    }
#else
    (void)data;
    (void)size;
    (void)port;
    (void)cbArg;
#endif
}

/* Rx Isr for non-gPTP traffic */
static void EnetApp_rxIsrFxn(void *appData)
{
    SemaphoreP_post(&gEnetAppCfg.rxSemObj);
}

static int32_t EnetApp_openDma()
{
    EnetApp_GetDmaHandleInArgs     txInArgs;
    EnetApp_GetTxDmaHandleOutArgs  txChInfo;
    int32_t status = ENET_SOK;

    /* Open the TX channel */
    txInArgs.cbArg   = NULL;
    txInArgs.notifyCb = NULL;

    EnetApp_getTxDmaHandle((ENET_DMA_TX_CH0),
                           &txInArgs,
                           &txChInfo);

    gEnetAppCfg.txChNum = txChInfo.txChNum;
    gEnetAppCfg.hTxCh   = txChInfo.hTxCh;

    if (gEnetAppCfg.hTxCh == NULL)
    {
#if FIX_RM
        /* Free the channel number if open Tx channel failed */
        EnetAppUtils_freeTxCh(gEnetAppCfg.hEnet,
                              gEnetAppCfg.coreKey,
                              gEnetAppCfg.coreId,
                              gEnetAppCfg.txChNum);
#endif
        EnetAppUtils_print("EnetApp_openDma() failed to open TX channel\r\n");
        status = ENET_EFAIL;
        EnetAppUtils_assert(gEnetAppCfg.hTxCh != NULL);
    }

    /* Allocate TX packets and keep them locally enqueued */
    if (status == ENET_SOK)
    {
        EnetApp_initTxFreePktQ();
    }

    /* Open the RX flow for Regular frames */
    if (status == ENET_SOK)
    {
        EnetApp_GetDmaHandleInArgs     rxInArgs;
        EnetApp_GetRxDmaHandleOutArgs  rxChInfo;

        rxInArgs.notifyCb = EnetApp_rxIsrFxn;
        rxInArgs.cbArg   = NULL;

        EnetApp_getRxDmaHandle(ENET_DMA_RX_CH0,
                               &rxInArgs,
                               &rxChInfo);
#if defined(ENET_SOC_HOSTPORT_DMA_TYPE_UDMA)
        gEnetAppCfg.rxStartFlowIdx = rxChInfo.rxFlowStartIdx;
        gEnetAppCfg.rxFlowIdx = rxChInfo.rxFlowIdx;
#else
        gEnetAppCfg.rxFlowIdx = rxChInfo.rxChNum;
#endif
        gEnetAppCfg.hRxCh  = rxChInfo.hRxCh;
        EnetAppUtils_assert(rxChInfo.numValidMacAddress == 1);
        EnetUtils_copyMacAddr(gEnetAppCfg.macAddr, rxChInfo.macAddr[rxChInfo.numValidMacAddress - 1]);
        EnetAppUtils_print("MAC port addr: ");
        EnetAppUtils_printMacAddr(gEnetAppCfg.macAddr);

        if (gEnetAppCfg.hRxCh == NULL)
        {
            EnetAppUtils_print("EnetApp_openRxCh() failed to open RX flow\r\n");
            status = ENET_EFAIL;
            EnetAppUtils_assert(gEnetAppCfg.hRxCh != NULL);
        }
    }

    /* Submit all ready RX buffers to DMA */
    if (status == ENET_SOK)
    {

        EnetApp_initRxReadyPktQ(gEnetAppCfg.hRxCh);
    }

     return status;
}

static void EnetApp_closeDma()
{
    EnetDma_PktQ fqPktInfoQ;
    EnetDma_PktQ cqPktInfoQ;

    EnetQueue_initQ(&fqPktInfoQ);
    EnetQueue_initQ(&cqPktInfoQ);

    /* Close Regular RX channel */
    EnetApp_closeRxDma(ENET_DMA_RX_CH0,
                       gEnetAppCfg.hEnet,
                       gEnetAppCfg.coreKey,
                       gEnetAppCfg.coreId,
                       &fqPktInfoQ,
                       &cqPktInfoQ);

    EnetAppUtils_freePktInfoQ(&fqPktInfoQ);
    EnetAppUtils_freePktInfoQ(&cqPktInfoQ);

    /* Close TX channel */
    EnetQueue_initQ(&fqPktInfoQ);
    EnetQueue_initQ(&cqPktInfoQ);

    /* Retrieve any pending TX packets from driver */
    EnetApp_retrieveFreeTxPkts();

    EnetApp_closeTxDma(ENET_DMA_TX_CH0,
                       gEnetAppCfg.hEnet,
                       gEnetAppCfg.coreKey,
                       gEnetAppCfg.coreId,
                       &fqPktInfoQ,
                       &cqPktInfoQ);

    EnetAppUtils_freePktInfoQ(&fqPktInfoQ);
    EnetAppUtils_freePktInfoQ(&cqPktInfoQ);

    EnetAppUtils_freePktInfoQ(&gEnetAppCfg.txFreePktInfoQ);
}

static void EnetApp_initTxFreePktQ(void)
{
    EnetDma_Pkt *pPktInfo;
    uint32_t i;
    uint32_t scatterSegments[] = { ENET_MEM_LARGE_POOL_PKT_SIZE };

    /* Initialize TX EthPkts and queue them to txFreePktInfoQ */
    for (i = 0U; i < ENET_DMA_TX_CH0_NUM_PKTS; i++)
    {
        pPktInfo = EnetMem_allocEthPkt(&gEnetAppCfg,
                                       ENETDMA_CACHELINE_ALIGNMENT,
                                       ENET_ARRAYSIZE(scatterSegments),
                                       scatterSegments);
        EnetAppUtils_assert(pPktInfo != NULL);
        ENET_UTILS_SET_PKT_APP_STATE(&pPktInfo->pktState, ENET_PKTSTATE_APP_WITH_FREEQ);

        EnetQueue_enq(&gEnetAppCfg.txFreePktInfoQ, &pPktInfo->node);
    }

    EnetAppUtils_print("initQs() txFreePktInfoQ initialized with %d pkts\r\n",
                       EnetQueue_getQCount(&gEnetAppCfg.txFreePktInfoQ));
}

static void EnetApp_initRxReadyPktQ(EnetDma_RxChHandle hRxCh)
{
    EnetDma_PktQ rxReadyQ;
    EnetDma_PktQ rxFreeQ;
    EnetDma_Pkt *pPktInfo;
    uint32_t i;
    int32_t status;
    uint32_t scatterSegments[] = { ENET_MEM_LARGE_POOL_PKT_SIZE };

    EnetQueue_initQ(&rxFreeQ);

    for (i = 0U; i < ENET_DMA_RX_CH0_NUM_PKTS; i++)
    {
        pPktInfo = EnetMem_allocEthPkt(&gEnetAppCfg,
                                       ENETDMA_CACHELINE_ALIGNMENT,
                                       ENET_ARRAYSIZE(scatterSegments),
                                       scatterSegments);
        EnetAppUtils_assert(pPktInfo != NULL);

        ENET_UTILS_SET_PKT_APP_STATE(&pPktInfo->pktState, ENET_PKTSTATE_APP_WITH_FREEQ);

        EnetQueue_enq(&rxFreeQ, &pPktInfo->node);
    }

    /* Retrieve any packets which are ready */
    EnetQueue_initQ(&rxReadyQ);
    status = EnetDma_retrieveRxPktQ(hRxCh, &rxReadyQ);
    EnetAppUtils_assert(status == ENET_SOK);

    /* There should not be any packet with DMA during init */
    EnetAppUtils_assert(EnetQueue_getQCount(&rxReadyQ) == 0U);

    EnetAppUtils_validatePacketState(&rxFreeQ,
                                     ENET_PKTSTATE_APP_WITH_FREEQ,
                                     ENET_PKTSTATE_APP_WITH_DRIVER);

    EnetDma_submitRxPktQ(hRxCh, &rxFreeQ);

    /* Assert here, as during init, the number of DMA descriptors should be equal to
     * the number of free Ethernet buffers available with app */
    EnetAppUtils_assert(EnetQueue_getQCount(&rxFreeQ) == 0U);
}

static uint32_t EnetApp_retrieveFreeTxPkts()
{
    EnetDma_PktQ txFreeQ;
    EnetDma_Pkt *pktInfo;
    uint32_t txFreeQCnt = 0U;
    int32_t status;

    EnetQueue_initQ(&txFreeQ);

    /* Retrieve any packets that may be free now */
    status = EnetDma_retrieveTxPktQ(gEnetAppCfg.hTxCh, &txFreeQ);
    if (status == ENET_SOK)
    {
        txFreeQCnt = EnetQueue_getQCount(&txFreeQ);

        pktInfo = (EnetDma_Pkt *)EnetQueue_deq(&txFreeQ);
        while (NULL != pktInfo)
        {
            EnetDma_checkPktState(&pktInfo->pktState,
                                    ENET_PKTSTATE_MODULE_APP,
                                    ENET_PKTSTATE_APP_WITH_DRIVER,
                                    ENET_PKTSTATE_APP_WITH_FREEQ);

            EnetQueue_enq(&gEnetAppCfg.txFreePktInfoQ, &pktInfo->node);
            pktInfo = (EnetDma_Pkt *)EnetQueue_deq(&txFreeQ);
        }
    }
    else
    {
        EnetAppUtils_print("retrieveFreeTxPkts() failed to retrieve pkts: %d\r\n", status);
    }

    return txFreeQCnt;
}

/* Rx Echo task for non-gPTP traffic */
static void EnetApp_rxTask(void *args)
{
    EnetDma_PktQ rxReadyQ;
    EnetDma_PktQ rxFreeQ;
    EnetDma_PktQ txSubmitQ;
    EnetDma_Pkt *rxPktInfo;
    EthFrame *rxFrame;
    uint32_t totalRxCnt = 0U;
    int32_t status = ENET_SOK;

    EnetAppUtils_print("%s: default RX flow started\r\n",
                       gEnetAppCfg.name);

    while ((ENET_SOK == status))
    {
        /* Wait for packet reception */
        SemaphoreP_pend(&gEnetAppCfg.rxSemObj, SystemP_WAIT_FOREVER);

        /* All peripherals have single hardware RX channel, so we only need to retrieve
         * packets from a single flow.*/
        EnetQueue_initQ(&rxReadyQ);
        EnetQueue_initQ(&rxFreeQ);
        EnetQueue_initQ(&txSubmitQ);

        /* Get the packets received so far */
        status = EnetDma_retrieveRxPktQ(gEnetAppCfg.hRxCh, &rxReadyQ);
        if (status != ENET_SOK)
        {
            /* Should we bail out here? */
            EnetAppUtils_print("Failed to retrieve RX pkt queue: %d\r\n", status);
            continue;
        }
#if DEBUG
        EnetAppUtils_print("%s: Received %u packets\r\n", gEnetAppCfg.name, EnetQueue_getQCount(&rxReadyQ));
#endif
        totalRxCnt += EnetQueue_getQCount(&rxReadyQ);

        /* Consume the received packets and send them back */
        rxPktInfo = (EnetDma_Pkt *)EnetQueue_deq(&rxReadyQ);
        while (rxPktInfo != NULL)
        {
            rxFrame = (EthFrame *)rxPktInfo->sgList.list[0].bufPtr;
            uint32_t isPhyStatus = 0U;
            uint32_t frameLen = rxPktInfo->sgList.list[0].segmentFilledLen;
#if ENETAPP_ENABLE_PTP_PATH_DEBUG
            static uint32_t dbgTotal = 0U;
            static uint32_t dbgPhyStatus = 0U;
            static uint32_t dbgPtpNonPsf = 0U;
            static uint32_t dbgOther = 0U;
            static uint32_t dbgPtpEchoed = 0U;
            static uint32_t dbgPtpDropped = 0U;
            static uint32_t dbgPtpNoTxPkt = 0U;
            static uint32_t dbgFirstNLogs = 0U;
            static uint32_t dbgLastPrintTimeMs = 0U;
            uint32_t dbgL2HdrLen = 0U;
            uint16_t dbgEthType = 0U;
            uint16_t dbgSeqId = 0xFFFFU;
            uint8_t dbgMsgType = 0xFFU;
            uint32_t isNonPsfPtp = 0U;
            const uint32_t nowMs = ClockP_getTimeUsec() / 1000U;
            const char *ptpAction = "n/a";
#endif
            EnetDma_checkPktState(&rxPktInfo->pktState,
                                    ENET_PKTSTATE_MODULE_APP,
                                    ENET_PKTSTATE_APP_WITH_DRIVER,
                                    ENET_PKTSTATE_APP_WITH_READYQ);

            {
                /* ======================= Phase-1 PHY PSF ingest ======================= */
                /*
                 * Forward every default-flow PHY status frame into the new PHY IOCTL.
                 * This keeps the active RX ingress point unchanged while validating
                 * the end-to-end API path into the PHY driver.
                 */
                static const uint8_t phyStatusHdr[14U] = {
                    0x01, 0x1B, 0x19, 0x00, 0x00, 0x00,
                    0x08, 0x00, 0x17, 0x0B, 0x6B, 0x0F,
                    0x88, 0xF7,
                };
                const uint8_t *psfFrame = (const uint8_t *)rxFrame;
                const uint32_t psfFrameLen = rxPktInfo->sgList.list[0].segmentFilledLen;

                if ((psfFrameLen >= sizeof(phyStatusHdr)) &&
                    (memcmp(psfFrame, phyStatusHdr, sizeof(phyStatusHdr)) == 0))
                {
                    isPhyStatus = 1U;
                    EnetPhy_PtpStatusFrameInArgs inArgs;
                    Enet_IoctlPrms psfPrms;
                    Enet_MacPort psfMacPort = ENET_MAC_PORT_1;
                    int32_t psfStatus = ENET_SOK;

                    if (gEnetAppCfg.numMacPorts > 0U)
                    {
                        psfMacPort = gEnetAppCfg.macPorts[0U];
                    }

                    inArgs.macPort = psfMacPort;
                    inArgs.frame = psfFrame;
                    inArgs.frameLen = psfFrameLen;
                    ENET_IOCTL_SET_IN_ARGS(&psfPrms, &inArgs);
                    ENET_IOCTL(gEnetAppCfg.hEnet,
                               gEnetAppCfg.coreId,
                               ENET_PHY_IOCTL_PTP_PROCESS_STATUS_FRAME,
                               &psfPrms,
                               psfStatus);

                    if (psfStatus != ENET_SOK)
                    {
                        EnetAppUtils_print("%s: PHY PSF ingest ioctl failed: %d\r\n",
                                           gEnetAppCfg.name,
                                           psfStatus);
                    }
                }
            }

#if ENETAPP_ENABLE_PTP_PATH_DEBUG
            dbgTotal++;
            if (isPhyStatus)
            {
                dbgPhyStatus++;
            }
            else
            {
                dbgEthType = EnetApp_getEthType((const uint8_t *)rxFrame, frameLen, &dbgL2HdrLen);
                if (dbgEthType == 0x88F7U)
                {
                    isNonPsfPtp = 1U;
                    dbgPtpNonPsf++;
                    dbgMsgType = EnetApp_getPtpMsgType((const uint8_t *)rxFrame,
                                                       frameLen,
                                                       dbgL2HdrLen,
                                                       &dbgSeqId);
                }
                else
                {
                    dbgOther++;
                }
            }
#endif

#if ENETAPP_ENABLE_RX_FRAME_DEBUG
            {
                /* ======================= PSF Debug: default RX flow ======================= */
                /*
                 * Periodic RX frame overview (single-block debug helper)
                 * Expected gPTP app traffic buckets:
                 *   - phyStatus : default TI PHY status frame header match
                 *   - ptp       : IEEE 1588/gPTP frames (0x88F7), with message-type split
                 *   - ipv4/arp/ipv6/lldp/tsn
                 *   - vlanTagged and other
                 */
                static const uint8_t phyStatusHdr[14U] = {
                    0x01, 0x1B, 0x19, 0x00, 0x00, 0x00,
                    0x08, 0x00, 0x17, 0x0B, 0x6B, 0x0F,
                    0x88, 0xF7,
                };
                static uint32_t total = 0U;
                static uint32_t phyStatus = 0U;
                static uint32_t ptp = 0U;
                static uint32_t ptpSync = 0U;
                static uint32_t ptpDelayReq = 0U;
                static uint32_t ptpPdelayReq = 0U;
                static uint32_t ptpPdelayResp = 0U;
                static uint32_t ptpFollowUp = 0U;
                static uint32_t ptpDelayResp = 0U;
                static uint32_t ptpPdelayRespFollowUp = 0U;
                static uint32_t ptpAnnounce = 0U;
                static uint32_t ptpSignaling = 0U;
                static uint32_t ptpManagement = 0U;
                static uint32_t ptpOtherMsg = 0U;
                static uint32_t ipv4 = 0U;
                static uint32_t arp = 0U;
                static uint32_t ipv6 = 0U;
                static uint32_t lldp = 0U;
                static uint32_t tsn = 0U;
                static uint32_t vlanTagged = 0U;
                static uint32_t other = 0U;
                static uint32_t lastPrintTimeMs = 0U;
                static uint32_t sampleDecimCounter = 0U;
#if ENETAPP_ENABLE_RX_FRAME_DUMP
                static uint32_t dumpDecimCounter = 0U;
                static uint32_t lastDumpTimeMs = 0U;
#endif
                uint16_t ethType = 0U;
                uint32_t l2HdrLen = 14U;
                uint32_t dumpBucket = 0U;
                const char *bucketName = "other";
                const uint8_t *frame = (const uint8_t *)rxFrame;
                const uint32_t frameLen = rxPktInfo->sgList.list[0].segmentFilledLen;
                const uint32_t nowMs = ClockP_getTimeUsec() / 1000U;
                uint32_t sampleEveryN = ENETAPP_RX_FRAME_DEBUG_SAMPLE_EVERY_N_FRAMES;

                if (sampleEveryN == 0U)
                {
                    sampleEveryN = 1U;
                }

                sampleDecimCounter++;
                if ((frameLen >= 14U) && (sampleDecimCounter >= sampleEveryN))
                {
                    sampleDecimCounter = 0U;
                    total++;

                    if (memcmp(frame, phyStatusHdr, sizeof(phyStatusHdr)) == 0)
                    {
                        phyStatus++;
                        dumpBucket = ENETAPP_RX_DUMP_BUCKET_PHY_STATUS;
                        bucketName = "phyStatus";
                    }
                    else
                    {
                        ethType = ((uint16_t)frame[12U] << 8U) | frame[13U];

                        if ((ethType == 0x8100U) || (ethType == 0x88A8U))
                        {
                            vlanTagged++;
                            dumpBucket = ENETAPP_RX_DUMP_BUCKET_VLAN;
                            bucketName = "vlan";
                            l2HdrLen += 4U;
                            if (frameLen >= l2HdrLen)
                            {
                                ethType = ((uint16_t)frame[l2HdrLen - 2U] << 8U) | frame[l2HdrLen - 1U];
                            }
                        }

                        if ((ethType == 0x8100U) || (ethType == 0x88A8U))
                        {
                            vlanTagged++;
                            dumpBucket = ENETAPP_RX_DUMP_BUCKET_VLAN;
                            bucketName = "vlan";
                            l2HdrLen += 4U;
                            if (frameLen >= l2HdrLen)
                            {
                                ethType = ((uint16_t)frame[l2HdrLen - 2U] << 8U) | frame[l2HdrLen - 1U];
                            }
                        }

                        if (ethType == 0x88F7U)
                        {
                            uint8_t msgType = 0xFFU;
                            ptp++;
                            dumpBucket = ENETAPP_RX_DUMP_BUCKET_PTP;
                            bucketName = "ptp";
                            if (frameLen > l2HdrLen)
                            {
                                msgType = frame[l2HdrLen] & 0x0FU;
                            }
                            switch (msgType)
                            {
                                case 0x0U: ptpSync++; break;
                                case 0x1U: ptpDelayReq++; break;
                                case 0x2U: ptpPdelayReq++; break;
                                case 0x3U: ptpPdelayResp++; break;
                                case 0x8U: ptpFollowUp++; break;
                                case 0x9U: ptpDelayResp++; break;
                                case 0xAU: ptpPdelayRespFollowUp++; break;
                                case 0xBU: ptpAnnounce++; break;
                                case 0xCU: ptpSignaling++; break;
                                case 0xDU: ptpManagement++; break;
                                default:   ptpOtherMsg++; break;
                            }
                        }
                        else if (ethType == 0x0800U)
                        {
                            ipv4++;
                            dumpBucket = ENETAPP_RX_DUMP_BUCKET_IPV4;
                            bucketName = "ipv4";
                        }
                        else if (ethType == 0x0806U)
                        {
                            arp++;
                        }
                        else if (ethType == 0x86DDU)
                        {
                            ipv6++;
                        }
                        else if (ethType == 0x88CCU)
                        {
                            lldp++;
                        }
                        else if (ethType == 0x22F0U)
                        {
                            tsn++;
                        }
                        else
                        {
                            other++;
                            dumpBucket = ENETAPP_RX_DUMP_BUCKET_OTHER;
                            bucketName = "other";
                        }
                    }

                    if (lastPrintTimeMs == 0U)
                    {
                        lastPrintTimeMs = nowMs;
                    }
                    else if ((nowMs - lastPrintTimeMs) >= ENETAPP_RX_FRAME_DEBUG_PRINT_INTERVAL_MS)
                    {
                        EnetAppUtils_print(
                            "%s: RX frame overview: total=%u phyStatus=%u ptp=%u ipv4=%u arp=%u ipv6=%u lldp=%u tsn=%u vlan=%u other=%u\r\n",
                            gEnetAppCfg.name,
                            total,
                            phyStatus,
                            ptp,
                            ipv4,
                            arp,
                            ipv6,
                            lldp,
                            tsn,
                            vlanTagged,
                            other);
                        EnetAppUtils_print(
                            "%s: RX ptp detail: sync=%u dreq=%u pdreq=%u pdresp=%u fup=%u dresp=%u pdfup=%u ann=%u sig=%u mgmt=%u oth=%u\r\n",
                            gEnetAppCfg.name,
                            ptpSync,
                            ptpDelayReq,
                            ptpPdelayReq,
                            ptpPdelayResp,
                            ptpFollowUp,
                            ptpDelayResp,
                            ptpPdelayRespFollowUp,
                            ptpAnnounce,
                            ptpSignaling,
                            ptpManagement,
                            ptpOtherMsg);
                        lastPrintTimeMs = nowMs;
                    }

#if ENETAPP_ENABLE_RX_FRAME_DUMP
                    {
                        uint32_t dumpDecimation = ENETAPP_RX_FRAME_DUMP_DECIMATION;
                        uint32_t maxDumpBytes = ENETAPP_RX_FRAME_DUMP_MAX_BYTES;
                        const uint32_t bytesPerLine = 16U;
                        uint32_t i;

                        if (dumpDecimation == 0U)
                        {
                            dumpDecimation = 1U;
                        }

                        dumpDecimCounter++;
                        if (maxDumpBytes > frameLen)
                        {
                            maxDumpBytes = frameLen;
                        }

                        if ((maxDumpBytes > 0U) &&
                            ((ENETAPP_RX_FRAME_DUMP_BUCKET_MASK & dumpBucket) != 0U) &&
                            (dumpDecimCounter >= dumpDecimation) &&
                            ((lastDumpTimeMs == 0U) || ((nowMs - lastDumpTimeMs) >= ENETAPP_RX_FRAME_DUMP_INTERVAL_MS)))
                        {
                            dumpDecimCounter = 0U;
                            lastDumpTimeMs = nowMs;

                            EnetAppUtils_print("%s: RX dump bucket=%s len=%u bytes=%u\r\n",
                                               gEnetAppCfg.name,
                                               bucketName,
                                               frameLen,
                                               maxDumpBytes);

                            for (i = 0U; i < maxDumpBytes; i++)
                            {
                                if ((i % bytesPerLine) == 0U)
                                {
                                    EnetAppUtils_print("  %03u:", i);
                                }

                                EnetAppUtils_print(" %02x", frame[i]);

                                if ((((i + 1U) % bytesPerLine) == 0U) || ((i + 1U) == maxDumpBytes))
                                {
                                    EnetAppUtils_print("\r\n");
                                }
                            }
                        }
                    }
#endif
                }
            }
#endif

#if ENETAPP_DROP_PHY_STATUS_FROM_ECHO
            if (isPhyStatus)
            {
#if ENETAPP_ENABLE_PTP_PATH_DEBUG
                ptpAction = "drop-phy-status";
#endif
                EnetDma_checkPktState(&rxPktInfo->pktState,
                                      ENET_PKTSTATE_MODULE_APP,
                                      ENET_PKTSTATE_APP_WITH_READYQ,
                                      ENET_PKTSTATE_APP_WITH_FREEQ);

                /* Keep PSFs in the RX path for ingest/debug, but do not echo
                 * them back onto the link through the example's generic echo
                 * transmit path. */
                EnetQueue_enq(&rxFreeQ, &rxPktInfo->node);
                rxPktInfo = (EnetDma_Pkt *)EnetQueue_deq(&rxReadyQ);
                continue;
            }
#endif

#if ENETAPP_ENABLE_DEFAULT_RX_ECHO
            /* Retrieve TX packets from driver and recycle them */
            EnetApp_retrieveFreeTxPkts();

            /* Dequeue one free TX Eth packet */
            {
                EnetDma_Pkt *txPktInfo;
                EthFrame *txFrame;

                txPktInfo = (EnetDma_Pkt *)EnetQueue_deq(&gEnetAppCfg.txFreePktInfoQ);
                if (txPktInfo != NULL)
                {
                    /* Fill the TX Eth frame with test content */
                    txFrame = (EthFrame *)txPktInfo->sgList.list[0].bufPtr;
                    memcpy(txFrame->hdr.dstMac, rxFrame->hdr.srcMac, ENET_MAC_ADDR_LEN);
                    memcpy(txFrame->hdr.srcMac, &gEnetAppCfg.macAddr[0U], ENET_MAC_ADDR_LEN);
                    txFrame->hdr.etherType = rxFrame->hdr.etherType;

                    txPktInfo->sgList.list[0].segmentFilledLen = rxPktInfo->sgList.list[0].segmentFilledLen;
                    EnetAppUtils_assert(txPktInfo->sgList.list[0].segmentAllocLen >= txPktInfo->sgList.list[0].segmentFilledLen);
                    memcpy(&txFrame->payload[0U],
                            &rxFrame->payload[0U],
                            rxPktInfo->sgList.list[0].segmentFilledLen - sizeof(EthFrameHeader));

                    txPktInfo->sgList.numScatterSegments = 1;
                    txPktInfo->chkSumInfo = 0U;
                    txPktInfo->appPriv = &gEnetAppCfg;
                    txPktInfo->tsInfo.enableHostTxTs = false;

                    EnetDma_checkPktState(&txPktInfo->pktState,
                                            ENET_PKTSTATE_MODULE_APP,
                                            ENET_PKTSTATE_APP_WITH_FREEQ,
                                            ENET_PKTSTATE_APP_WITH_DRIVER);

                    /* Enqueue the packet for later transmission */
                    EnetQueue_enq(&txSubmitQ, &txPktInfo->node);
#if ENETAPP_ENABLE_PTP_PATH_DEBUG
                    if (isNonPsfPtp)
                    {
                        dbgPtpEchoed++;
                        ptpAction = "echo";
                    }
#endif
                }
                else
                {
                    EnetAppUtils_print("%s: Drop due to TX pkt not available\r\n", gEnetAppCfg.name);
#if ENETAPP_ENABLE_PTP_PATH_DEBUG
                    if (isNonPsfPtp)
                    {
                        dbgPtpNoTxPkt++;
                        ptpAction = "drop-no-tx-pkt";
                    }
#endif
                }
            }
#else
#if ENETAPP_ENABLE_PTP_PATH_DEBUG
            if (isNonPsfPtp)
            {
                dbgPtpDropped++;
                ptpAction = "drop-echo-disabled";
            }
#endif
#endif

#if ENETAPP_ENABLE_PTP_PATH_DEBUG
            if (isNonPsfPtp && (dbgFirstNLogs < ENETAPP_PTP_PATH_DEBUG_FIRST_N))
            {
                dbgFirstNLogs++;
                EnetAppUtils_print(
                    "%s: default-flow PTP[%u] len=%u dst=%02x:%02x:%02x:%02x:%02x:%02x src=%02x:%02x:%02x:%02x:%02x:%02x type=%s(0x%02x) seqId=%u action=%s\r\n",
                    gEnetAppCfg.name,
                    dbgFirstNLogs,
                    frameLen,
                    rxFrame->hdr.dstMac[0U], rxFrame->hdr.dstMac[1U], rxFrame->hdr.dstMac[2U],
                    rxFrame->hdr.dstMac[3U], rxFrame->hdr.dstMac[4U], rxFrame->hdr.dstMac[5U],
                    rxFrame->hdr.srcMac[0U], rxFrame->hdr.srcMac[1U], rxFrame->hdr.srcMac[2U],
                    rxFrame->hdr.srcMac[3U], rxFrame->hdr.srcMac[4U], rxFrame->hdr.srcMac[5U],
                    EnetApp_getPtpMsgTypeName(dbgMsgType),
                    dbgMsgType,
                    dbgSeqId,
                    ptpAction);
            }

            if (dbgLastPrintTimeMs == 0U)
            {
                dbgLastPrintTimeMs = nowMs;
            }
            else if ((nowMs - dbgLastPrintTimeMs) >= ENETAPP_RX_FRAME_DEBUG_PRINT_INTERVAL_MS)
            {
                EnetAppUtils_print(
                    "%s: default-flow summary: total=%u phyStatus=%u ptpNonPsf=%u other=%u ptpEchoed=%u ptpDropped=%u ptpNoTx=%u echoEnabled=%u\r\n",
                    gEnetAppCfg.name,
                    dbgTotal,
                    dbgPhyStatus,
                    dbgPtpNonPsf,
                    dbgOther,
                    dbgPtpEchoed,
                    dbgPtpDropped,
                    dbgPtpNoTxPkt,
                    ENETAPP_ENABLE_DEFAULT_RX_ECHO);
                dbgLastPrintTimeMs = nowMs;
            }
#endif

            EnetDma_checkPktState(&rxPktInfo->pktState,
                                    ENET_PKTSTATE_MODULE_APP,
                                    ENET_PKTSTATE_APP_WITH_READYQ,
                                    ENET_PKTSTATE_APP_WITH_FREEQ);

            /* Release the received packet */
            EnetQueue_enq(&rxFreeQ, &rxPktInfo->node);
            rxPktInfo = (EnetDma_Pkt *)EnetQueue_deq(&rxReadyQ);
        }

        EnetApp_printRxDropStats();

#if ENETAPP_ENABLE_DEFAULT_RX_ECHO
        /* Transmit all enqueued packets */
        status = EnetDma_submitTxPktQ(gEnetAppCfg.hTxCh, &txSubmitQ);
        if (status != ENET_SOK)
        {
            EnetAppUtils_print("%s: Failed to submit TX pkt queue: %d\r\n", gEnetAppCfg.name, status);
        }
#endif

        EnetAppUtils_validatePacketState(&rxFreeQ,
                                            ENET_PKTSTATE_APP_WITH_FREEQ,
                                            ENET_PKTSTATE_APP_WITH_DRIVER);

        /* Submit now processed buffers */
        EnetDma_submitRxPktQ(gEnetAppCfg.hRxCh, &rxFreeQ);
        if (status != ENET_SOK)
        {
            EnetAppUtils_print("%s: Failed to submit RX pkt queue: %d\r\n", gEnetAppCfg.name, status);
        }
    }

    EnetAppUtils_print("%s: Received %u packets\r\n", gEnetAppCfg.name, totalRxCnt);

    TaskP_exit();
}

void EnetApp_createRxTask()
{
    TaskP_Params taskParams;
    int32_t status;

    status = EnetApp_openDma();
    DebugP_assert(ENET_SOK == status);

    status = SemaphoreP_constructBinary(&gEnetAppCfg.rxSemObj, 0);
    DebugP_assert(SystemP_SUCCESS == status);
    TaskP_Params_init(&taskParams);
    taskParams.priority       = 2U;
    taskParams.stack          = gEnetAppTaskStackRx;
    taskParams.stackSize      = sizeof(gEnetAppTaskStackRx);
    taskParams.args           = (void*)&gEnetAppCfg;
    taskParams.name           = "Rx Task";
    taskParams.taskMain       = &EnetApp_rxTask;

    status = TaskP_construct(&gEnetAppCfg.rxTaskObj, &taskParams);
    DebugP_assert(SystemP_SUCCESS == status);
}

void EnetApp_destroyRxTask()
{
    SemaphoreP_destruct(&gEnetAppCfg.rxSemObj);
    TaskP_destruct(&gEnetAppCfg.rxTaskObj);
    EnetApp_closeDma();
}
